    def rob(self, nums: List[int]) -> int:
        dp = [0] * len(nums)
        if not nums:
            return 0
        elif len(nums) == 1:
            return nums[0]
        for i in range(1, len(nums)):
            if i == 1:
                dp[1] = max(nums[0], nums[1])
            else:
                dp[i] = max(dp[i-1], nums[i] + dp[i - 2])
        
class Solution:
        
        dp[0] = nums[0]
        return dp[-1]
