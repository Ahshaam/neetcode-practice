    def remove(self, val: int) -> bool:
        if val not in self.idx:
            return False
        i = self.idx[val]
        last = self.vals[-1]
        self.vals[i] = last          # 
        move last element into the hole
        self.idx[last] = i           # 
        update its recorded position
        self.vals.pop()              # O
        (1), removes from the end
        del self.idx[val]
        return True
